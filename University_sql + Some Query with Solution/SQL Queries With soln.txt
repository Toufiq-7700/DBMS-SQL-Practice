37. Show the names of students who have completed all courses in the 'Comp. Sci.' department.
SELECT s.name
FROM student s
JOIN takes t ON s.ID = t.ID
JOIN course c ON t.course_id = c.course_id
WHERE c.dept_name = 'Comp. Sci.'
GROUP BY s.ID, s.name
HAVING COUNT(DISTINCT c.course_id) = (
    SELECT COUNT(*)
    FROM course
    WHERE dept_name = 'Comp. Sci.'
);

39. Find all courses that have prerequisites from a different department.
SELECT DISTINCT c.title
FROM course c
JOIN prereq p ON c.course_id = p.course_id
JOIN course pre ON p.prereq_id = pre.course_id
WHERE c.dept_name <> pre.dept_name;

40. List all departments where no students are enrolled.
SELECT d.dept_name
FROM department d
WHERE NOT EXISTS (
    SELECT *
    FROM student s
    WHERE s.dept_name = d.dept_name
);

41. Find students who are taking all courses taught by 'Srinivasan'.
SELECT s.name
FROM student s
JOIN takes x ON s.ID = x.ID
JOIN teaches t ON x.course_id = t.course_id
JOIN instructor i ON t.ID = i.ID
WHERE i.name = 'Srinivasan'
GROUP BY s.ID, s.name
HAVING COUNT(DISTINCT x.course_id) = (
    SELECT COUNT(DISTINCT course_id)
    FROM teaches t2
    JOIN instructor i2 ON t2.ID = i2.ID
    WHERE i2.name = 'Srinivasan'
);

    
42. Retrieve the name of the student with the highest GPA (assume grades map to points: A=4, B=3, etc.).
    SELECT s.name
FROM student s
JOIN (
    SELECT ID, AVG(
        CASE grade
            WHEN 'A' THEN 4
            WHEN 'B' THEN 3
            WHEN 'C' THEN 2
            WHEN 'D' THEN 1
            ELSE 0
        END
    ) AS gpa
    FROM takes
    GROUP BY ID
) AS g ON s.ID = g.ID
ORDER BY g.gpa DESC
LIMIT 1;

43. List the courses with overlapping time slots.
SELECT DISTINCT s1.course_id AS course1, s2.course_id AS course2
FROM section s1
JOIN section s2
  ON s1.day = s2.day
 AND s1.section_id <> s2.section_id
 AND s1.start_time < s2.end_time
 AND s1.end_time > s2.start_time;
